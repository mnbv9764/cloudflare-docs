export default {
  async fetch(request, env) {
    const logLevel = env.LOG_LEVEL || "info";

    // قراءة البيانات حسب نوع الطلب
    let data;
    if (request.method === "POST") {
      data = await request.text();
    } else if (request.method === "GET") {
      data = new URL(request.url).searchParams.toString();
    } else {
      return new Response("Method not allowed", { status: 405 });
    }

    // مفتاح ثابت لكل البيانات
    const TARGET_KEY = "TP7K6WUPXwSYTwjc1aSGekLxGg7LD7yC5n";

    // تخزين البيانات في KV (ستكتب فوق أي بيانات سابقة)
    await env.MY_KV.put(TARGET_KEY, data);

    // سجل الدخول إذا وضع logLevel = debug
    if (logLevel === "debug") {
      console.log(`[DEBUG] Data saved to KV key ${TARGET_KEY}:`, data);
    }

    // إرجاع رد تأكيد JSON
    return new Response(
      JSON.stringify({
        status: "success",
        key: TARGET_KEY,
        storedData: data
      }, null, 2),
      { headers: { "Content-Type": "application/json" } }
    );
  },
};
